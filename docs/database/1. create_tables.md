# English

This command will create all the tables required to run correctly the service on Supabase.

# Portuguese

Esse comando ira criar todas as tabelas necessarias para executar a aplicação corretamente com o Supabase.

</br></br></br>

```roomsql
-- Create Table: user
CREATE TABLE IF NOT EXISTS users (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    name text NOT NULL,
    email text NOT NULL,
    avatar_url text
);
COMMENT ON TABLE users IS 'This table is related with the users of the app';

-- Create Table: financial_categories
CREATE TABLE IF NOT EXISTS financial_categories (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    name text NOT NULL,
    icon text NOT NULL,
    icon_color text NOT NULL DEFAULT '#000',
    owner text NOT NULL,
    owner_id uuid NOT NULL,
    parent_id uuid,
    depth_level int2,
    type text NOT NULL DEFAULT 'EXPENSE'
);
ALTER TABLE financial_categories ADD COLUMN IF NOT EXISTS type text NOT NULL DEFAULT 'EXPENSE';
COMMENT ON TABLE financial_categories IS 'This table is related with the financial categories of the user or family';

-- Create Table: family
CREATE TABLE IF NOT EXISTS families (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    family_name text NOT NULL,
    owner_id uuid NOT NULL
);
COMMENT ON TABLE families IS 'This table is related with the family of the user, each user can have multiple families';

-- Create Table: financial_transactions
CREATE TABLE IF NOT EXISTS financial_transactions (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    description text NOT NULL,
    date text NOT NULL,
    value int8 NOT NULL,
    category text NOT NULL,
    category_id uuid NOT NULL REFERENCES financial_categories(id) ON DELETE CASCADE,
    account_id uuid NOT NULL REFERENCES financial_accounts(id) ON DELETE CASCADE,
    type text NOT NULL,
    owner text NOT NULL,
    owner_id uuid NOT NULL
);
COMMENT ON TABLE financial_transactions IS 'This table is related with the financial transactions of the user or family';

-- Create Table: storage
CREATE TABLE IF NOT EXISTS storage_items (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    owner_id uuid NOT NULL,
    owner text NOT NULL,
    description text NOT NULL,
    quantity int4 NOT NULL,
    unit text NOT NULL,
    purchase_date timestamp with time zone,
    expiration_date timestamp with time zone,
    barcode text,
    brand text,
    notes text,
    opening_date timestamp with time zone
);
COMMENT ON TABLE storage_items IS 'This table is related with the storage items of the user or family';

-- Create Table: financial_accounts
CREATE TABLE IF NOT EXISTS financial_accounts (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    name text NOT NULL,
    icon text NOT NULL,
    owner text NOT NULL,
    owner_id uuid NOT NULL,
    status text NOT NULL,
    balance int8 NOT NULL
);
COMMENT ON TABLE financial_accounts IS 'This table is related with the financial accounts of the user or family';
```

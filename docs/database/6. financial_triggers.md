Here is the triggered required to update the information when something related with storage happen.

## Delete in cascate

```roomsql
CREATE OR REPLACE FUNCTION cascade_delete_financial()
RETURNS TRIGGER AS $$
BEGIN
    IF (TG_TABLE_NAME = 'families') THEN
        DELETE FROM financial_transactions WHERE owner = 'FAMILY' AND owner_id = OLD.id;
        DELETE FROM financial_categories WHERE owner = 'FAMILY' AND owner_id = OLD.id;
        DELETE FROM financial_accounts WHERE owner = 'FAMILY' AND owner_id = OLD.id;
    ELSIF (TG_TABLE_NAME = 'users') THEN
        DELETE FROM financial_transactions WHERE owner = 'USER' AND owner_id = OLD.id;
        DELETE FROM financial_categories WHERE owner = 'USER' AND owner_id = OLD.id;
        DELETE FROM financial_accounts WHERE owner = 'USER' AND owner_id = OLD.id;
    ELSIF (TG_TABLE_NAME = 'financial_categories') THEN
        DELETE FROM financial_transactions WHERE category_id = OLD.id;
    ELSIF (TG_TABLE_NAME = 'financial_accounts') THEN
        DELETE FROM financial_transactions WHERE account_id = OLD.id;
    END IF;
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

-- Attach the trigger to families table
DROP TRIGGER IF EXISTS trigger_cascade_delete_financial_on_families ON families;
CREATE TRIGGER trigger_cascade_delete_financial_on_families
AFTER DELETE ON families
FOR EACH ROW
EXECUTE FUNCTION cascade_delete_financial();

-- Attach the trigger to users table
DROP TRIGGER IF EXISTS trigger_cascade_delete_financial_on_users ON users;
CREATE TRIGGER trigger_cascade_delete_financial_on_users
AFTER DELETE ON users
FOR EACH ROW
EXECUTE FUNCTION cascade_delete_financial();

-- Attach the trigger to financial_categories table
DROP TRIGGER IF EXISTS trigger_cascade_delete_financial_on_categories ON financial_categories;
CREATE TRIGGER trigger_cascade_delete_financial_on_categories
AFTER DELETE ON financial_categories
FOR EACH ROW
EXECUTE FUNCTION cascade_delete_financial();

-- Attach the trigger to financial_accounts table
DROP TRIGGER IF EXISTS trigger_cascade_delete_financial_on_accounts ON financial_accounts;
CREATE TRIGGER trigger_cascade_delete_financial_on_accounts
AFTER DELETE ON financial_accounts
FOR EACH ROW
EXECUTE FUNCTION cascade_delete_financial();
```
